# --- web layer -------------------------------------------------------------
Flask==3.0.3
flask-cors==4.0.1
pymongo==4.8.0

# --- ML inference runtime (freight forecast models) ------------------------
# The pickles in app/ml_models/ were trained on scikit-learn 1.9.0; loading them
# under a different minor version raises an InconsistentVersionWarning and can
# fail outright, so the version is pinned rather than floated.
numpy>=2.1,<3
scipy>=1.13
joblib>=1.4
scikit-learn==1.9.0
pandas>=2.2

# --- live market data pipeline (ported from ML/data.py, ML/src/run_all_data.py)
# yfinance pulls the same three daily series the models were trained on —
# BDRY (freight), BZ=F (Brent) and USDINR=X — so the API forecasts from today's
# market instead of the frozen training CSV. openpyxl reads the World Bank
# monthly "Pink Sheet" .xlsx that ML/wordbank.py ingests.
yfinance>=0.2.40
openpyxl>=3.1.2

# --- live port congestion (ported from ML/app.py get_live_port_congestion) ---
requests>=2.31
beautifulsoup4>=4.12
lxml>=5.0

# --- sourcing optimiser ----------------------------------------------------
# ML/app.py states the allocation as a PuLP linear program; steel_engine solves
# the identical program and falls back to its closed form if PuLP is absent.
PuLP>=2.8
